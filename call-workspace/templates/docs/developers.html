    Autenticación

    Todos los endpoints usan autenticación por sesión Django. Debes hacer login primero y conservar la cookie sessionid en todas las peticiones.

    Paso 1 — Obtener CSRF token y hacer login
    import requests

BASE_URL = "http://localhost:8001"
session = requests.Session()

# Obtener CSRF token
csrf_response = session.get(f"{BASE_URL}/accounts/login/")
csrf_token = csrf_response.cookies["csrftoken"]

# Hacer login
login = session.post(
    f"{BASE_URL}/accounts/login/",
    data={
        "username": "tu_usuario",
        "password": "tu_contraseña",
        "csrfmiddlewaretoken": csrf_token,
    },
)
print("Login OK:", login.ok)  # True si fue exitoso
    Paso 2 — Usar la sesión en todas las peticiones
    # El objeto `session` conserva las cookies automáticamente.
# Siempre pasa el header X-CSRFToken en POST/PUT/DELETE:

headers = {"X-CSRFToken": session.cookies.get("csrftoken", "")}

response = session.post(url, json=payload, headers=headers)
print(response.json())
    Swagger interactivo: Disponible en http://localhost:8001/api/v1/docs — permite probar todos los endpoints directamente desde el navegador.

    Lotes — Múltiples llamadas

    Envía un arreglo de llamadas en un solo request. Todas quedan en cola y se procesan de forma asíncrona.

    POST /api/v1/batch/ — Crear lote de llamadas 
    headers = {"X-CSRFToken": session.cookies.get("csrftoken", "")}

payload = {
    "campaign_id": 1,    # ID de la campaña (debe tener script y estar activa)
    "calls": [
        {
            "phone_number": "+56912345678",
            "input_params": {
                "nombre": "Juan García",
                "consecionario": "Chery Norte",
                "fecha_agenda": "15/05/2026"
            }
        },
        {
            "phone_number": "+56987654321",
            "input_params": {
                "nombre": "María López",
                "consecionario": "Chery Sur",
                "fecha_agenda": "16/05/2026"
            }
        },
    ]
}

response = session.post(f"{BASE_URL}/api/v1/batch/", json=payload, headers=headers)
result = response.json()
print(result)
# {"batch_job_id": 3, "total_calls": 2, "status": "pending"}
    Crear lote desde archivo CSV

    El CSV debe tener columna phone_number y una columna por cada variable del script.

    # Formato del CSV:
# phone_number,nombre,consecionario,fecha_agenda
# +56912345678,Juan García,Chery Norte,15/05/2026
# +56987654321,María López,Chery Sur,16/05/2026

import pandas as pd

df = pd.DataFrame([
    {
        "phone_number": "+56912345678",
        "nombre": "Juan García",
        "consecionario": "Chery Norte",
        "fecha_agenda": "15/05/2026"
    },
    {
        "phone_number": "+56987654321",
        "nombre": "María López",
        "consecionario": "Chery Sur",
        "fecha_agenda": "16/05/2026"
    }
])

# Convertir a formato de la API
calls = []
for _, row in df.iterrows():
    phone = row.pop("phone_number")
    calls.append({"phone_number": phone, "input_params": row.to_dict()})

response = session.post(
    f"{BASE_URL}/api/v1/batch/",
    json={"campaign_id": 1, "calls": calls},
    headers={"X-CSRFToken": session.cookies.get("csrftoken", "")},
)
print(response.json())
    Monitorear progreso del lote
    import time

batch_job_id = result["batch_job_id"]

while True:
    status = session.get(f"{BASE_URL}/api/v1/batch/{batch_job_id}/").json()
    total  = status["total_calls"]
    done   = status["done_calls"]
    failed = status["failed_calls"]

    print(f"Progreso: {done + failed}/{total} — {status['status']}")

    if status["status"] in ("completed", "failed"):
        break

    time.sleep(5)
    Respuestas de error:
    • 400 — Campaña sin script asignado
    • 400 — Faltan input_params requeridos por el script
    • 404 — campaign_id no existe o está inactiva
    Llamada individual

    Para lanzar una sola llamada usa el endpoint de batch con un arreglo de un elemento.

    headers = {"X-CSRFToken": session.cookies.get("csrftoken", "")}

response = session.post(
    f"{BASE_URL}/api/v1/batch/",
    json={
        "campaign_id": 1,
        "calls": [
            {
                "phone_number": "+56912345678",
                "input_params": {
                    "nombre": "Pedro Soto",
                    "consecionario": "Chery Norte",
                    "fecha_agenda": "20/05/2026"
                }
            }
        ]
    },
    headers=headers,
)
result = response.json()
print("Batch ID:", result["batch_job_id"])

# Esperar resultado
import time
while True:
    job = session.get(f"{BASE_URL}/api/v1/batch/{result['batch_job_id']}/").json()
    if job["status"] in ("completed", "failed"):
        break
    time.sleep(3)

# Ver el resultado de la llamada
calls = session.get(
    f"{BASE_URL}/api/v1/calls/",
    params={"campaign_id": 1, "status": "done"}
).json()

for call in calls:
    analysis = session.get(f"{BASE_URL}/api/v1/calls/{call['id']}/analysis/").json()
    print(f"{call['phone_number']} → {analysis['output_data']}")
    # output_data: {"confirmacion": "asistira"}
    Los resultados ([[output_params]] del script) quedan disponibles en analysis.output_data una vez que la llamada está en estado done.

    Exportar todos los resultados de una campaña
    import pandas as pd

calls = session.get(
    f"{BASE_URL}/api/v1/calls/",
    params={"campaign_id": 1, "status": "done"}
).json()

rows = []
for call in calls:
    analysis_resp = session.get(f"{BASE_URL}/api/v1/calls/{call['id']}/analysis/")
    if analysis_resp.status_code == 200:
        analysis = analysis_resp.json()
        rows.append({
            "phone_number": call["phone_number"],
            "status": call["status"],
            **analysis["output_data"],   # expande confirmacion, etc.
            "summary": analysis["summary"],
            "score": analysis["compliance_score"],
        })

df = pd.DataFrame(rows)
df.to_csv("resultados_campana.csv", index=False)
print(df)
    Scripts — Sintaxis y variables

    Los scripts definen el comportamiento del bot. Tienen 3 tipos de marcadores:

    {{variable}}

    Variable de entrada. Se reemplaza antes de la llamada con los datos del batch (input_params). También puede ir en el saludo.

    [[variable]]

    Variable de salida. El bot la captura durante la conversación. Queda disponible en analysis.output_data.

    ## comentario ##

    Nota interna solo visible al crear el script. El bot no la lee ni la dice.

    Ejemplo de script completo
    Saludo:
  Hola {{nombre}}, te llamamos desde {{consecionario}}.

Prompt:
  Contacta a {{nombre}} para confirmar su cita del {{fecha_agenda}}
  en {{consecionario}}. Pregunta si asistirá.

  ## Opciones válidas: asistira, re-agenda, cancela ##

  Registra la respuesta en:
  [[confirmacion]]
    Relación script → batch → resultado
    # El script detecta automáticamente:
# input_params  = ["nombre", "consecionario", "fecha_agenda"]
# output_params = ["confirmacion"]

# En el batch debes proveer TODOS los input_params:
"input_params": {
    "nombre": "Juan García",          # reemplaza {{nombre}}
    "consecionario": "Chery Norte",   # reemplaza {{consecionario}}
    "fecha_agenda": "15/05/2026"      # reemplaza {{fecha_agenda}}
}

# Después de la llamada, el resultado queda en:
analysis["output_data"] == {"confirmacion": "asistira"}
    Si envías un batch sin todos los input_params que requiere el script, la API retorna 400: Missing input params.
    Obtener variables requeridas de un script por código
    script = session.get(f"{BASE_URL}/scripts/api/1/json/").json()

print("Inputs requeridos:", script["input_params"])
# ["nombre", "consecionario", "fecha_agenda"]

print("Outputs capturados:", script["output_params"])
# ["confirmacion"]

    Snippet de sesión reutilizable
    import requests

class VoiceBotClient:
    def __init__(self, base_url, username, password):
        self.base = base_url
        self.session = requests.Session()
        self._login(username, password)

    def _login(self, username, password):
        r = self.session.get(f"{self.base}/accounts/login/")
        self.session.post(
            f"{self.base}/accounts/login/",
            data={
                "username": username,
                "password": password,
                "csrfmiddlewaretoken": r.cookies["csrftoken"],
            },
        )

    def _headers(self):
        return {"X-CSRFToken": self.session.cookies.get("csrftoken", "")}

    def create_batch(self, campaign_id, calls):
        return self.session.post(
            f"{self.base}/api/v1/batch/",
            json={"campaign_id": campaign_id, "calls": calls},
            headers=self._headers(),
        ).json()

    def get_results(self, campaign_id):
        calls = self.session.get(
            f"{self.base}/api/v1/calls/",
            params={"campaign_id": campaign_id, "status": "done"},
        ).json()
        results = []
        for call in calls:
            r = self.session.get(f"{self.base}/api/v1/calls/{call['id']}/analysis/")
            if r.status_code == 200:
                results.append({**call, "analysis": r.json()})
        return results


# Uso:
client = VoiceBotClient("http://localhost:8001", "admin", "admin123")

batch = client.create_batch(1, [
    {"phone_number": "+56912345678", "input_params": {"nombre": "Juan", "consecionario": "Chery", "fecha_agenda": "20/05"}},
])
print(batch)

results = client.get_results(1)
for r in results:
    print(r["phone_number"], r["analysis"]["output_data"])
